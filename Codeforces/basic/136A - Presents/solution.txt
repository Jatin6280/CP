#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    cin >> n;

    vector<int> p(n + 1);
    vector<int> ans(n + 1);

    // Read input
    for (int i = 1; i <= n; i++) {
        cin >> p[i];
    }

    // Reverse the mapping
    for (int i = 1; i <= n; i++) {
        ans[p[i]] = i;
    }

    // Print answer
    for (int i = 1; i <= n; i++) {
        cout << ans[i] << " ";
    }

    return 0;
}
